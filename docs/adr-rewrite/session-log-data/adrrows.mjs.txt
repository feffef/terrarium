import { readFileSync } from 'node:fs'
const d = JSON.parse(readFileSync('./data.json','utf8')).sort((a,b)=>a.startedAt.localeCompare(b.startedAt))
const bn = p => p.replace(/.*\.claude\/rules\//,'')
for (const s of d) {
  const seen = new Set(); const rows=[]
  for (const x of s.docsRead) { const m = x.path.match(/^docs\/adr\/(\d{4})/); if (!m) continue; if (seen.has(m[1])) continue; seen.add(m[1])
    const n=m[1]; const shown=[...s.rulesGlobal,...s.rulesLoaded].map(bn).some(r=>r.startsWith('adr-'+n)); const fu=s.rulesFollowedUp.map(bn).some(r=>r.startsWith('adr-'+n))
    rows.push(`${n} ${shown?'S':'-'}${fu?'F':'-'} ${x.useful??'none'} :: ${x.reason}`) }
  if (rows.length) console.log(`\n# ${s.file.match(/session_(\w{7})/)[1]} ${s.startedAt.slice(0,10)} ${s.kind} ${s.hasRV?'':'EARLY'} | ${s.goal}\n`+rows.join('\n'))
}
